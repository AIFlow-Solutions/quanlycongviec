Cài Đặt "Bưu Tá" Gửi Mã OTP Thật Về Hộp Thư Email

Cài 1 lần duy nhất — Tự động gửi mã 6 số vào Inbox của Sếp & 4 Nhân viên

💡 Cơ chế hoạt động: Bạn dùng 1 tài khoản Gmail làm "Bưu tá phát mã" thông qua công cụ miễn phí Google Apps Script của chính Google. Khi Sếp hoặc bất kỳ nhân viên nào nhập Email của họ vào màn hình đăng nhập, Google sẽ tự động gửi 1 bức thư chứa 6 số OTP tới thẳng hộp thư của người đó!
BƯỚC 1: Sao chép đoạn mã "Bưu tá Google" bên dưới 
/**
 * Google Apps Script Web App - WorkFlow Pro
 * Chức năng:
 * 1. Gửi email chứa mã OTP 6 chữ số đến hộp thư người dùng
 * 2. Lưu trữ & Đồng bộ dữ liệu toàn hệ thống (Cloud Database cho Quản lý & Nhân viên)
 */

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput(JSON.stringify({ ok: false, error: "No post data" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    var data = JSON.parse(e.postData.contents);

    // 1. GỬI MÃ OTP ĐĂNG NHẬP QUA EMAIL
    if (data.action === "send_otp" || (data.otp && !data.action)) {
      var recipient = data.email || (e.parameter ? e.parameter.email : "");
      var otpCode = data.otp || (e.parameter ? e.parameter.otp : "");
      if (recipient && otpCode) {
        MailApp.sendEmail({
          to: recipient,
          name: "Hệ Thống Quản Lý Hồ Sơ (No-Reply)",
          noReply: true,
          subject: "[Hệ Thống Hồ Sơ] Mã xác thực OTP đăng nhập: " + otpCode,
          htmlBody: "<div style='font-family:sans-serif;padding:20px;border:1px solid #ddd;border-radius:12px;max-width:480px'>"
            + "<h2 style='color:#9e0b22;margin-top:0'>Xác thực Đăng nhập Hệ thống Hồ sơ</h2>"
            + "<p>Xin chào <b>" + recipient + "</b>,</p>"
            + "<p>Mã xác thực OTP (6 chữ số) để truy cập bảng công việc của bạn là:</p>"
            + "<div style='font-size:28px;font-weight:bold;letter-spacing:6px;color:#1d4ed8;background:#eff6ff;padding:12px 20px;text-align:center;border-radius:8px;margin:16px 0'>" + otpCode + "</div>"
            + "<p style='color:#666;font-size:12px'>Đây là email tự động từ hệ thống, vui lòng không phản hồi.</p></div>"
        });
      }
      return ContentService.createTextOutput(JSON.stringify({ ok: true, action: "send_otp" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 2. LƯU DỮ LIỆU ĐỒNG BỘ TOÀN HỆ THỐNG (CLOUD SYNC SAVE)
    if (data.action === "sync_save") {
      saveToProperties(data.payload || "{}");
      return ContentService.createTextOutput(JSON.stringify({ ok: true, action: "sync_save", time: new Date().toISOString() }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 3. TẢI DỮ LIỆU ĐỒNG BỘ MỚI NHẤT (CLOUD SYNC LOAD)
    if (data.action === "sync_load") {
      var raw = loadFromProperties();
      return ContentService.createTextOutput(JSON.stringify({ ok: true, action: "sync_load", payload: raw || null }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  var raw = loadFromProperties();
  return ContentService.createTextOutput(JSON.stringify({ ok: true, payload: raw || null }))
    .setMimeType(ContentService.MimeType.JSON);
}

function saveToProperties(jsonStr) {
  var props = PropertiesService.getScriptProperties();
  var chunkSize = 8000;
  var count = Math.ceil(jsonStr.length / chunkSize);
  var oldKeys = props.getKeys();
  for (var i = 0; i < oldKeys.length; i++) {
    if (oldKeys[i].indexOf("CHUNK_") === 0) {
      props.deleteProperty(oldKeys[i]);
    }
  }
  props.setProperty("CHUNK_COUNT", count.toString());
  for (var i = 0; i < count; i++) {
    props.setProperty("CHUNK_" + i, jsonStr.substring(i * chunkSize, (i + 1) * chunkSize));
  }
}

function loadFromProperties() {
  var props = PropertiesService.getScriptProperties();
  var countStr = props.getProperty("CHUNK_COUNT");
  if (!countStr) return null;
  var count = parseInt(countStr, 10);
  var str = "";
  for (var i = 0; i < count; i++) {
    str += props.getProperty("CHUNK_" + i) || "";
  }
  return str;
}
BƯỚC 2: Dán vào Google Script & Lấy đường link (Chỉ mất 30 giây)
1. Bấm mở script.google.com → Nhấn nút Dự án mới (New project) ở góc trái.
2. Xóa chữ có sẵn, Dán (Ctrl+V) đoạn mã vừa sao chép ở Bước 1 vào.
3. Nhấn nút màu xanh Triển khai (Deploy) ở góc trên bên phải → Chọn Tùy chọn triển khai mới (New deployment).
4. Bấm hình bánh răng ⚙️ chọn loại: Ứng dụng web (Web app) → Phần "Ai có quyền truy cập (Who has access)" chọn: Bất kỳ ai (Anyone) → Nhấn Triển khai (Deploy) & cấp quyền Gmail.
5. Copy đường link URL ứng dụng web (`https://script.google.com/macros/s/.../exec`) dán vào Bước 3 bên dưới.

Sau khi lưu, bất kỳ ai (Sếp hoặc 4 nhân viên) nhập email của họ sẽ nhận được thư OTP thật trong Inbox, đồng thời mã hiển thị thử nghiệm trên màn hình sẽ tự động ẩn đi!

Khai Báo Danh Sách Email Được Phép Truy Cập (Whitelist)

🔒 Quy tắc bảo mật: Chỉ những địa chỉ Email có trong bảng dưới đây mới nhận được mã OTP để đăng nhập.

  • Email khai báo là 👨‍💻 Nhân viên: Khi đăng nhập chỉ vào được bảng cá nhân của chính mình (khóa hoàn toàn quyền xem của Sếp và nhân viên khác).
